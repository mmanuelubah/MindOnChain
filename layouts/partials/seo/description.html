line. */ -}} {{- $d := "" -}} {{- $title := partial "display-title.html" . -}} {{- if .IsHome -}} {{- $d = site.Params.description -}} {{- else if .Description -}} {{- $d = .Description -}} {{- else (ne .Params.ecu "Unknown")) (printf " for the %s" .Params.ecu) "" -}} {{- $d = printf "%s: %sprogramming notes%s from the Alltech Solutions workshop in Lagos, with vehicle and ECU details and
